283 double skaiciuotiVidurki() const {
284 if (paz_.empty()) return 0.0;
285 double suma = 0.0;
286 for (int p : paz_) suma += p;
287 return suma / static_cast<double>(paz_.size());
288 }
289
294 double skaiciuotiMediana() const {
295 if (paz_.empty()) return 0.0;
296 vector<int> tmp = paz_;
297 sort(tmp.begin(), tmp.end());
298 size_t n = tmp.size();
299 if (n % 2 == 0) return (tmp[n / 2 - 1] + tmp[n / 2]) / 2.0;
300 return tmp[n / 2];
301 }
302
309 void apskaiciuoti(int metodas) {
310 if (metodas == 1 || metodas == 3)
311 gal_vid_ = skaiciuotiVidurki() * 0.4 + egz_ * 0.6;
312 if (metodas == 2 || metodas == 3)
313 gal_med_ = skaiciuotiMediana() * 0.4 + egz_ * 0.6;
314 }

341inline istream& operator>>(istream& is, Studentas& st) {
342 string eilute;
343 if (!getline(is, eilute)) return is;
344 if (eilute.empty()) return is;
345
346 stringstream ss(eilute);
347 string v, p;
348 if (!(ss >> v >> p)) return is;
349
350 st.vardas_ = v;
351 st.pavarde_ = p;
352 st.paz_.clear();
353 st.egz_ = 0;
354 st.gal_vid_ = 0.0;
355 st.gal_med_ = 0.0;
356
357 int n;
358 while (ss >> n) st.paz_.push_back(n);
359 if (!st.paz_.empty()) {
360 st.egz_ = st.paz_.back();
361 st.paz_.pop_back();
362 }
363 return is;
364}
